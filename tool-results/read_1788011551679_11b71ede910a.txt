   134→function formatBytes(bytes: number | null | undefined): string {
   135→  if (bytes == null || Number.isNaN(bytes)) return '未知大小';
   136→  if (bytes < 1024) return `${bytes} B`;
   137→  const units = ['KB', 'MB', 'GB', 'TB'];
   138→  let v = bytes / 1024;
   139→  let i = 0;
   140→  while (v >= 1024 && i < units.length - 1) {
   141→    v /= 1024;
   142→    i++;
   143→  }
   144→  return `${v.toFixed(v >= 100 ? 0 : v >= 10 ? 1 : 2)} ${units[i]}`;
   145→}
   146→
   147→function timeAgo(iso: string): string {
   148→  const diff = Date.now() - new Date(iso).getTime();
   149→  const m = Math.floor(diff / 60000);
   150→  if (m < 1) return '刚刚';
   151→  if (m < 60) return `${m} 分钟前`;
   152→  const h = Math.floor(m / 60);
   153→  if (h < 24) return `${h} 小时前`;
   154→  return `${Math.floor(h / 24)} 天前`;
   155→}

   157→async function copyText(text: string, tip = '已复制到剪贴板') {
   158→  try {
   159→    await navigator.clipboard.writeText(text);
   160→    toast.success(tip);
   161→  } catch {
   162→    toast.error('复制失败,请手动复制');
   163→  }
   164→}
