   129→function formatBytes(bytes: number | null | undefined): string {
   130→  if (bytes == null || Number.isNaN(bytes)) return '未知大小';
   131→  if (bytes < 1024) return `${bytes} B`;
   132→  const units = ['KB', 'MB', 'GB', 'TB'];
   133→  let v = bytes / 1024;
   134→  let i = 0;
   135→  while (v >= 1024 && i < units.length - 1) {
   136→    v /= 1024;
   137→    i++;
   138→  }
   139→  return `${v.toFixed(v >= 100 ? 0 : v >= 10 ? 1 : 2)} ${units[i]}`;
   140→}
   141→
   142→function timeAgo(iso: string): string {
   143→  const diff = Date.now() - new Date(iso).getTime();
   144→  const m = Math.floor(diff / 60000);
   145→  if (m < 1) return '刚刚';
   146→  if (m < 60) return `${m} 分钟前`;
   147→  const h = Math.floor(m / 60);
   148→  if (h < 24) return `${h} 小时前`;
   149→  return `${Math.floor(h / 24)} 天前`;
   150→}

   152→async function copyText(text: string, tip = '已复制到剪贴板') {
   153→  try {
   154→    await navigator.clipboard.writeText(text);
   155→    toast.success(tip);
   156→  } catch {
   157→    toast.error('复制失败,请手动复制');
   158→  }
   159→}
