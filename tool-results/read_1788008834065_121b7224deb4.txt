   129→   129→function formatBytes(bytes: number | null | undefined): string {
   130→   130→  if (bytes == null || Number.isNaN(bytes)) return '未知大小';
   131→   131→  if (bytes < 1024) return `${bytes} B`;
   132→   132→  const units = ['KB', 'MB', 'GB', 'TB'];
   133→   133→  let v = bytes / 1024;
   134→   134→  let i = 0;
   135→   135→  while (v >= 1024 && i < units.length - 1) {
   136→   136→    v /= 1024;
   137→   137→    i++;
   138→   138→  }
   139→   139→  return `${v.toFixed(v >= 100 ? 0 : v >= 10 ? 1 : 2)} ${units[i]}`;
   140→   140→}

   142→   142→function timeAgo(iso: string): string {
   143→   143→  const diff = Date.now() - new Date(iso).getTime();
   144→   144→  const m = Math.floor(diff / 60000);
   145→   145→  if (m < 1) return '刚刚';
   146→   146→  if (m < 60) return `${m} 分钟前`;
   147→   147→  const h = Math.floor(m / 60);
   148→   148→  if (h < 24) return `${h} 小时前`;
   149→   149→  return `${Math.floor(h / 24)} 天前`;
   150→   150→}
